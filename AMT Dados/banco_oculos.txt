CREATE DATABASE bd_oculos_loja;

USE bd_oculos_loja;

CREATE TABLE cliente(
    id_cliente INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    nome_cliente VARCHAR(100),
    endereco_cliente VARCHAR(100),
    telefone_cliente CHAR(11)
);
CREATE TABLE armacao(
    id_armacao INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    tipo_armacao VARCHAR(30),
    cor_armacao VARCHAR(30)
);
CREATE TABLE receita_oculos(
    id_receita_oculos INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    id_cliente INT,
    data_emissao DATE,
    data_validade DATE,
    olho_direito_esferico DECIMAL(10,2),
    olho_direito_cilindrico DECIMAL(10,2),
    olho_direito_eixo DECIMAL(10,2),
    olho_direito_adicao DECIMAL(10,2),
    olho_direito_dnp DECIMAL(10,2),
    olho_esquerda_esferico DECIMAL(10,2),
    olho_esquerda_cilindrico DECIMAL(10,2),
    olho_esquerda_eixo DECIMAL(10,2),
    olho_esquerda_adicao DECIMAL(10,2),
    olho_esquerda_dnp DECIMAL(10,2),
    FOREIGN KEY (id_cliente) REFERENCES cliente(id_cliente)
);
CREATE TABLE entrega(
    id_entrega INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    id_cliente INT,
    id_armacao INT,
    id_receita_oculos INT,
    FOREIGN KEY (id_cliente) REFERENCES cliente(id_cliente),
    FOREIGN KEY (id_armacao) REFERENCES armacao(id_armacao),
    FOREIGN KEY (id_receita_oculos) REFERENCES receita_oculos(id_receita_oculos)
);
