CREATE DATABASE bd_viagens;
USE bd_viagens;

CREATE TABLE clientes(
    id_cli int PRIMARY KEY NOT NULL,
    nome VARCHAR(50),
    datanascimento DATE,
    endereco VARCHAR(80),
    email VARCHAR(50),
    cpf INT
);
CREATE TABLE destino(
    id_des INT PRIMARY KEY NOT NULL,
    destino_saida VARCHAR(50),
    cidade VARCHAR(30),
    estado VARCHAR(30),
    pais VARCHAR(30),
    preco_base DECIMAL(10,2)
);
CREATE TABLE viagens(
    id_via INT PRIMARY KEY NOT NULL,
    valor_viagen DECIMAL(10,2),
    id_des INT,
    destino VARCHAR(50),
    FOREIGN KEY (id_des) REFERENCES destino(id_des)
);
CREATE TABLE reserva(
    id_res INT PRIMARY KEY NOT NULL,
    id_via INT,
    id_cli INT,
    quantidade_pessoas INT,
    data_reserva DATE,
    status_reserva VARCHAR(30),
    FOREIGN KEY (id_via) REFERENCES viagens(id_via),
    FOREIGN KEY (id_cli) REFERENCES clientes(id_cli)
);
CREATE TABLE pagamento(
    id_pag INT PRIMARY KEY NOT NULL,
    id_res INT,
    id_cli INT,
    valor DECIMAL(10,2),
    data_pagamento DATE,
    status_pagamento VARCHAR(30),
    FOREIGN KEY (id_res) REFERENCES reserva(id_res),
    FOREIGN KEY (id_cli) REFERENCES clientes(id_cli)
);
