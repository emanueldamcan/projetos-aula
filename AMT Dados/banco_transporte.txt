CREATE DATABASE bd_transportador_catarinense;

USE bd_transportador_catarinense;

CREATE TABLE clientes(
    id_cliente INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    nome_cliente VARCHAR(100),
    endereco_cliente VARCHAR(100),
    cidade_cliente VARCHAR(100),
    telefone_cliente CHAR(11)
);
CREATE TABLE motoristas(
    id_motorista INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    nome_motorista VARCHAR(100),
    cnh_motorista VARCHAR(20),
    telefone_motorista CHAR(11)
);
CREATE TABLE veiculo(
    id_veiculo INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    placa_veiculo CHAR(7),
    modelo_veiculo VARCHAR(50),
    capacidade_kg_veiculo DECIMAL(10,2)
);
CREATE TABLE entregas(
    id_entrega INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    detalhes_entrega VARCHAR(150),
    peso_kg_entrega DECIMAL(10,2),
    id_cliente INT,
    id_motorista INT,
    id_veiculo INT,
    FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente),
    FOREIGN KEY (id_motorista) REFERENCES motoristas(id_motorista),
    FOREIGN KEY (id_veiculo) REFERENCES veiculo(id_veiculo)
);
